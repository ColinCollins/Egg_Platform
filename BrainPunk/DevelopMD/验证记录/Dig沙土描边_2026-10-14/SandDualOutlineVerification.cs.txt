    static Texture2D Render(Texture2D source, Material material, string name)
    {
        var rt = new RenderTexture(source.width, source.height, 0, RenderTextureFormat.ARGB32, RenderTextureReadWrite.Linear);
        rt.Create();
        // 外扩顶点需要真实相机的屏幕参数，使用SpriteRenderer而非没有有效投影的Blit夹具。
        var sprite = Sprite.Create(source,new Rect(0,0,source.width,source.height),new Vector2(0.5f,0.5f),128,0,SpriteMeshType.FullRect);
        var go = new GameObject("描边像素夹具");var sr=go.AddComponent<SpriteRenderer>();sr.sprite=sprite;sr.sharedMaterial=material;
        if(material.HasProperty("_SpriteLocalRect"))material.SetVector("_SpriteLocalRect",new Vector4(sprite.bounds.min.x,sprite.bounds.min.y,sprite.bounds.size.x,sprite.bounds.size.y));
        var cameraObject=new GameObject("描边像素相机");var camera=cameraObject.AddComponent<Camera>();camera.orthographic=true;camera.orthographicSize=source.height/256f;camera.transform.position=new Vector3(0,0,-10);camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.clear;camera.targetTexture=rt;camera.allowHDR=false;camera.allowMSAA=false;camera.Render();
        var previous = RenderTexture.active; RenderTexture.active = rt;
        var result = new Texture2D(rt.width, rt.height, TextureFormat.RGBA32, false, true);
        result.ReadPixels(new Rect(0,0,rt.width,rt.height), 0, 0); result.Apply();
        RenderTexture.active = previous;
        File.WriteAllBytes(Output + "dual-" + name + ".png", result.EncodeToPNG());
        rt.Release(); UnityEngine.Object.DestroyImmediate(rt);
        UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(cameraObject);UnityEngine.Object.DestroyImmediate(sprite);
        return result;
    }
